from bluesky import RunEngine
from bluesky.plans import scan
import bluesky.plan_stubs as bps
import bluesky.preprocessors as bpp
from ophyd import EpicsMotor, AreaDetector, SingleTrigger
from databroker import Broker

# Device definitions
rotation_motor = EpicsMotor('XF:28IDC{Rot:1-Ax:1}Mtr', name='rotation_motor')

class PhotogrammetryDetector(SingleTrigger, AreaDetector):
    pass

detector = PhotogrammetryDetector('XF:28IDC{Det:1}', name='photogrammetry_detector')

# Plan definition
@bpp.stage_decorator([rotation_motor, detector])
@bpp.run_decorator(md={'plan_name': 'photogrammetry_rotation_scan'})
def photogrammetry_rotation_scan(detector, rotation_motor, start_angle=0, stop_angle=180, 
                                num_projections=10, filename='test_file_1'):
    """
    Photogrammetry scan plan that rotates sample and captures images at each position.
    
    Parameters
    ----------
    detector : ophyd.Device
        The detector/camera to capture images
    rotation_motor : ophyd.Device  
        The rotation motor to move the sample
    start_angle : float, optional
        Starting angle in degrees (default: 0)
    stop_angle : float, optional
        Ending angle in degrees (default: 180) 
    num_projections : int, optional
        Number of projection images to capture (default: 10)
    filename : str, optional
        Base filename for saving data (default: 'test_file_1')
    """
    
    # Set metadata for the scan
    yield from bps.open_run(md={
        'scan_type': 'photogrammetry',
        'start_angle': start_angle,
        'stop_angle': stop_angle, 
        'num_projections': num_projections,
        'filename': filename,
        'angular_step': (stop_angle - start_angle) / (num_projections - 1)
    })
    
    # Configure detector for file saving
    yield from bps.abs_set(detector.cam.file_name, filename)
    yield from bps.abs_set(detector.cam.file_number, 1)
    yield from bps.abs_set(detector.cam.auto_increment, 1)
    
    # Perform the rotation scan
    yield from scan([detector], rotation_motor, start_angle, stop_angle, num_projections)
    
    yield from bps.close_run()
